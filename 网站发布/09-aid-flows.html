外援的去向与下行 — 谁在为脆弱国家买单？

2000 年外援占撒哈拉以南非洲卫生支出的 18%；2023 年只剩 9.4%。 但对最脆弱国家而言，外援仍是医疗最后的生命线。

10.1 9.1 ext 占 CHE 的世界版图

Code
if (!is.null(world_sf)) {
  d <- master_enriched |>
    dplyr::filter(.data$year == 2022, is.finite(.data$ext_che))
  map_df <- world_sf |> dplyr::left_join(d, by = "iso3_code")

  ggplot2::ggplot(map_df) +
    ggplot2::geom_sf(ggplot2::aes(fill = .data$ext_che),
                      colour = "white", linewidth = 0.15) +
    ggplot2::scale_fill_gradientn(
      colours = brand_palette$sequential,
      na.value = "grey90",
      labels = function(x) paste0(x, "%"),
      limits = c(0, 50), oob = scales::squish
    ) +
    ggplot2::guides(fill = ggplot2::guide_colorbar(
      title = "ext / CHE", barwidth = 14, barheight = 0.4)) +
    ggplot2::coord_sf(crs = "+proj=robin") +
    labs_news(
      title    = "\u5916\u63f4\u4f9d\u8d56\u4e16\u754c\u7248\u56fe \u00b7 2022",
      subtitle = "\u989c\u8272\u8d8a\u6df1\uff0c\u5916\u63f4\u5360\u603b\u5065\u5eb7\u5f00\u652f\u6bd4\u4f8b\u8d8a\u9ad8",
      x = NULL, y = NULL
    ) +
    ggplot2::theme(panel.grid.major = ggplot2::element_line(colour = "grey90"),
                   axis.text = ggplot2::element_blank())
}

10.2 9.2 ext 23 年下降轨迹（按收入组）

Code
d <- master_enriched |>
  dplyr::filter(is.finite(.data$ext_che), !is.na(.data$income_group)) |>
  dplyr::group_by(.data$year, .data$income_group) |>
  dplyr::summarise(median_ext = stats::median(.data$ext_che, na.rm = TRUE),
                   .groups = "drop")

ggplot2::ggplot(d, ggplot2::aes(.data$year, .data$median_ext,
                                  colour = .data$income_group)) +
  ggplot2::geom_line(linewidth = 1.4) +
  ggplot2::geom_point(size = 1.6) +
  scale_colour_brand_income() +
  ggplot2::scale_y_continuous(labels = function(x) paste0(x, "%")) +
  labs_news(
    title    = "\u5916\u63f4\u5728\u4f4e\u6536\u5165\u56fd\u4e0d\u4ec5\u6ca1\u51cf\u300d\u5728 2018 \u540e\u53cd\u800c\u8df3\u6da8",
    subtitle = "\u4e2d\u7b49\u6536\u5165\u56fd\u5219\u5448\u73b0\u7531 11% \u5230 4% \u7684\u7ed3\u6784\u6027\u4e0b\u964d",
    x = NULL, y = "ext / CHE (%)"
  )

10.3 9.3 高依赖国家 Top 15

Code
master_enriched |>
  dplyr::filter(.data$year == 2022, is.finite(.data$ext_che)) |>
  dplyr::arrange(dplyr::desc(.data$ext_che)) |>
  dplyr::slice_head(n = 15) |>
  dplyr::transmute(
    `国家` = .data$country_name,
    `大洲` = .data$continent,
    `ext (% CHE)` = round(.data$ext_che, 1),
    `总 CHE (亿 USD)` = round(.data$che_usd2023 / 1e9, 2)
  ) |>
  knitr::kable(caption = "\u5916\u63f4\u4f9d\u8d56\u5ea6\u6700\u9ad8 15 \u56fd \u00b7 2022")
