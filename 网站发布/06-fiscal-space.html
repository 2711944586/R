政府能撑多久？— 卫生支出占财政、债务、与 GDP

卫生支出不是凭空发生的 —— 它在政府的总预算里挤占别的开支，又被 GDP 总量与债务利息天花板限制。 本章看的是财政可持续性：哪些国家正在透支未来？

7.1 6.1 卫生占财政（GHE/GGE）的 23 年走势

Code
master_panel <- build_master_panel(master_enriched, external_data)

if ("gov_exp_gdp" %in% names(master_panel)) {
  d <- master_panel |>
    dplyr::filter(is.finite(.data$gghed_che),
                  is.finite(.data$gov_exp_gdp),
                  .data$gov_exp_gdp > 0) |>
    dplyr::mutate(ghe_gge = .data$gghed_che / .data$gov_exp_gdp * 100) |>
    dplyr::group_by(.data$year, .data$continent) |>
    dplyr::summarise(median_pct = stats::median(.data$ghe_gge,
                                                  na.rm = TRUE),
                     .groups = "drop") |>
    dplyr::filter(!is.na(.data$continent))

  ggplot2::ggplot(d, ggplot2::aes(.data$year, .data$median_pct,
                                    colour = .data$continent)) +
    ggplot2::geom_line(linewidth = 1.4) +
    scale_colour_brand_continent() +
    labs_news(
      title    = "\u536b\u751f\u5360\u653f\u5e9c\u603b\u652f\u51fa\u6bd4 \u00b7 \u5404\u5927\u6d32\u4e2d\u4f4d\u6570",
      subtitle = "\u4e1a\u9886\u533b\u4fdd\u4f53\u7cfb\u7684\u6b27\u6d32\u8fbe\u5230 \u224815%\uff0c\u9ad8\u4e8e\u5176\u4ed6\u5927\u6d32 5pp",
      x = NULL, y = "GHE/GGE (%)",
      caption = "\u6570\u636e\u6e90 \u00b7 GHED + IMF GFS  \u00b7  \u5206\u6790 Analysis: \u5e84\u9882"
    )
}

7.3 6.3 5 国轨迹：卫生 vs 债务

Code
focus5 <- c("USA", "JPN", "DEU", "CHN", "BRA")
if ("gov_debt_gdp" %in% names(master_panel)) {
  d <- master_panel |>
    dplyr::filter(.data$iso3_code %in% focus5,
                  is.finite(.data$gov_debt_gdp),
                  is.finite(.data$gghed_che))

  ggplot2::ggplot(d, ggplot2::aes(.data$year)) +
    ggplot2::geom_line(ggplot2::aes(y = .data$gov_debt_gdp,
                                      colour = "\u503a\u52a1/GDP"),
                       linewidth = 1) +
    ggplot2::geom_line(ggplot2::aes(y = .data$gghed_che * 5,
                                      colour = "GGHE-D / CHE \u00d7 5"),
                       linewidth = 1) +
    ggplot2::scale_y_continuous(
      name = "\u503a\u52a1 / GDP (%)",
      sec.axis = ggplot2::sec_axis(~ . / 5,
                                    name = "GGHE-D / CHE (%)")
    ) +
    ggplot2::scale_colour_manual(
      name = NULL,
      values = c("\u503a\u52a1/GDP" = brand_palette$source[["pvtd"]],
                 "GGHE-D / CHE \u00d7 5" = brand_palette$source[["gghed"]])
    ) +
    ggplot2::facet_wrap(~ .data$country_name, ncol = 3) +
    labs_news(
      title    = "\u4e94\u56fd\u8f68\u8ff9 \u00b7 \u503a\u52a1\u4e0e\u536b\u751f\u4e0d\u5fc5\u540c\u8c03",
      subtitle = "\u65e5\u672c\u9ad8\u503a\u52a1\u8fd8\u80fd\u7ef4\u6301\u9ad8\u653f\u5e9c\u533b\u4fdd\uff1b\u5371\u9669\u662f\u4e2d\u4f4e\u6536\u5165\u56fd",
      x = NULL, y = NULL
    )
}
Warning

主权风险信号：当政府债务 / GDP 超过 90% 同时 GGHE-D / CHE 下降 时，常预示着 IMF 紧缩条款下的 “crowding-out” 现象 —— 卫生支出会先于教育、福利被裁。斯里兰卡、阿根廷、加纳 在 2022–2023 都出现了类似信号。
