2030 之路 · 蒙特卡罗 1000 次模拟

如果中等收入国家在 2030 年前把 CHE/GDP 拉到 7%（OECD 中位水平）， 多少健康年 能被买到？— 我们用 1000 次蒙特卡罗给出概率云。

12.1 11.1 ARIMA 5 国预测（2024-2028）

Code
isos <- c("CHN", "USA", "IND", "BRA", "NGA")
preds <- lapply(isos, function(iso) {
  di <- master_enriched |>
    dplyr::filter(.data$iso3_code == iso) |> dplyr::arrange(.data$year)
  fc <- fit_forecast(di$che_pc_usd2023, di$year, h = 5)
  if (!is.null(fc)) {
    fc$iso3_code <- iso
    fc$country_name <- di$country_name[1]
    fc$historical <- FALSE
    hist <- tibble::tibble(
      iso3_code = iso, country_name = di$country_name[1],
      year = di$year, point = di$che_pc_usd2023,
      lo_80 = NA, hi_80 = NA, lo_95 = NA, hi_95 = NA,
      method = "Historical", historical = TRUE
    )
    dplyr::bind_rows(hist, fc)
  }
})
preds_df <- dplyr::bind_rows(Filter(Negate(is.null), preds))

if (nrow(preds_df)) {
  ggplot2::ggplot(preds_df, ggplot2::aes(.data$year, .data$point)) +
    ggplot2::geom_ribbon(ggplot2::aes(ymin = .data$lo_95,
                                        ymax = .data$hi_95),
                         fill = brand_palette$source[["gghed"]],
                         alpha = 0.18) +
    ggplot2::geom_ribbon(ggplot2::aes(ymin = .data$lo_80,
                                        ymax = .data$hi_80),
                         fill = brand_palette$source[["gghed"]],
                         alpha = 0.30) +
    ggplot2::geom_line(linewidth = 0.95,
                       colour = brand_palette$source[["gghed"]]) +
    ggplot2::facet_wrap(~ .data$country_name, scales = "free_y", ncol = 3) +
    ggplot2::scale_y_continuous(labels = scales::label_dollar()) +
    labs_news(
      title    = "\u4e94\u56fd\u4eba\u5747 CHE \u00b7 ARIMA(80 / 95% PI)",
      subtitle = "2024-2028 \u9884\u6d4b \u00b7 \u9634\u5f71\u8868\u793a\u4e0d\u786e\u5b9a\u6027\u5e26\u5bbd",
      x = NULL, y = "\u4eba\u5747 CHE (USD)"
    )
}

12.2 11.2 蒙特卡罗：把 CHE/GDP 推到 7%

Code
mc <- mc_scenarios(build_master_panel(master_enriched, external_data),
                    target_pct = 0.07, n_sims = 1000, horizon = 7)

if (!is.null(mc) && nrow(mc$summary)) {
  top <- mc$summary |>
    dplyr::arrange(dplyr::desc(.data$mean_gain)) |>
    dplyr::slice_head(n = 20) |>
    dplyr::mutate(country_name = forcats::fct_reorder(
      .data$country_name, .data$mean_gain))

  ggplot2::ggplot(top, ggplot2::aes(y = .data$country_name)) +
    ggplot2::geom_segment(ggplot2::aes(x = .data$lo95, xend = .data$hi95,
                                         yend = .data$country_name),
                          colour = "#1A1A1F22", linewidth = 4) +
    ggplot2::geom_segment(ggplot2::aes(x = .data$lo80, xend = .data$hi80,
                                         yend = .data$country_name),
                          colour = brand_palette$source[["gghed"]],
                          linewidth = 6) +
    ggplot2::geom_point(ggplot2::aes(x = .data$mean_gain),
                        size = 3, colour = "#1A1A1F") +
    ggplot2::scale_x_continuous(labels = function(x) paste0("+", round(x, 1), " yr")) +
    labs_news(
      title    = "\u8bbe\u60f3\u60c5\u666f \u00b7 \u4eba\u5747\u5bff\u547d\u9884\u671f\u589e\u76ca",
      subtitle = sprintf("1000 \u6b21\u8499\u7279\u5361\u6d1b \u00b7 horizon = %d \u5e74 \u00b7 \u5f39\u6027 = %.2f",
                          mc$horizon, mc$elasticity),
      x = "\u9884\u671f\u5bff\u547d\u589e\u76ca (\u5e74)", y = NULL
    )
}
Warning

模型局限：MC 假设了线性弹性与国家间共享系数，未考虑制度差异、医疗服务质量、人口结构变迁。结果应理解为 “如果其他条件不变” 的对冲假设，而不是真实预测。
