钱买到了健康吗？— 寿命、儿童死亡率与 UHC

医疗支出最终的考核标准不是”花了多少钱”，而是”换来多少健康”。 本章用 HALE / U5MR / UHC SCI 三把尺子，量化钱与健康之间的弹性。

9.1 8.1 弹性曲线：CHE/cap → 寿命

Code
master_panel <- build_master_panel(master_enriched, external_data)

if ("life_exp" %in% names(master_panel)) {
  d <- master_panel |>
    dplyr::filter(.data$year == 2021,
                  is.finite(.data$che_pc_usd2023), .data$che_pc_usd2023 > 0,
                  is.finite(.data$life_exp))

  ggplot2::ggplot(d, ggplot2::aes(.data$che_pc_usd2023, .data$life_exp,
                                    colour = .data$continent)) +
    ggplot2::geom_point(alpha = 0.65, size = 2) +
    ggplot2::geom_smooth(ggplot2::aes(group = 1),
                         method = "loess", span = 0.6, se = TRUE,
                         colour = "#1A1A1F", linewidth = 0.7) +
    ggplot2::scale_x_continuous(trans = "log10",
                                 labels = scales::label_dollar()) +
    scale_colour_brand_continent() +
    labs_news(
      title    = "\u4eba\u5747 CHE \u00d7 \u5bff\u547d \u00b7 \u201c\u62d0\u70b9\u201d\u5728 \u224810\u00b3 \u7f8e\u5143",
      subtitle = "\u8fc7\u4e86\u4e00\u4e2a\u9608\u503c\uff0c\u8d2d\u4e70\u5bff\u547d\u7684\u8fb9\u9645\u6536\u76ca\u5feb\u901f\u9012\u51cf",
      x = "\u4eba\u5747 CHE (USD, log)", y = "\u9884\u671f\u5bff\u547d (\u5e74)"
    )
}

9.2 8.2 分段弹性

Code
es <- fit_elasticity_segments(master_panel)
if (!is.null(es)) {
  es$fits |> dplyr::transmute(
    `区间`   = .data$seg,
    `样本 n` = .data$n,
    `弹性 β`   = round(.data$beta, 3),
    `SE`              = round(.data$se, 3),
    `R²`         = round(.data$r2, 3),
    `95% CI`          = sprintf("[%.2f, %.2f]", .data$ci_lo, .data$ci_hi)
  ) |>
  knitr::kable(caption = "\u5206\u6bb5\u5f39\u6027\uff1alog(CHE/cap) \u2192 log(\u5bff\u547d)")
}

9.3 8.3 UHC 服务覆盖指数 vs 卫生支出

if ("UHC_SCI" %in% names(master_panel)) {
  d <- master_panel |>
    dplyr::filter(is.finite(.data$UHC_SCI),
                  is.finite(.data$gghed_che),
                  .data$year %in% c(2017, 2019, 2021))
  ggplot2::ggplot(d, ggplot2::aes(.data$gghed_che, .data$UHC_SCI,
                                    colour = .data$continent)) +

      title    = "\u653f\u5e9c\u4e3b\u5bfc\u8f83\u9ad8\u8005 \u00b7 UHC \u670d\u52a1\u8986\u76d6\u8d8a\u5b8c\u5907",
